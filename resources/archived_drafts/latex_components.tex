% ALGORITHM 1: Cyclic Consensus Protocol
% ============================================================================
% Uses algorithm2e package for professional typesetting

\begin{algorithm}[t]
\caption{Cyclic Consensus Protocol}
\label{alg:consensus}
\DontPrintSemicolon
\SetAlgoLined
\SetKwInOut{Input}{Input}
\SetKwInOut{Output}{Output}
\SetKwFunction{ComputeScore}{ComputeScore}
\SetKwFunction{AllApproved}{AllApproved}
\SetKwFunction{HasHardRejection}{HasHardRejection}
\SetKwFunction{ExtractFeedback}{ExtractFeedback}
\SetKwFunction{Revise}{Revise}

\Input{Proposal $p$, Threshold $\theta$, Max iterations $k$}
\Output{ConsensusResult $\in \{\textsc{Approved}, \textsc{Rejected}, \textsc{NeedsReview}\}$}
\BlankLine
\For{$\text{iteration} \leftarrow 1$ \KwTo $k$}{
    $v_{\text{arch}} \leftarrow \text{Archeologist.verify}(p)$\;
    $v_{\text{arct}} \leftarrow \text{Architect.verify}(p)$\;
    $v_{\text{val}} \leftarrow \text{Validator.verify}(p)$\;
    \BlankLine
    $\text{score} \leftarrow \ComputeScore{v_{\text{arch}}, v_{\text{arct}}, v_{\text{val}}}$\;
    \BlankLine
    \uIf{$\text{score} \geq \theta$ \textbf{and} \AllApproved{$v_*$}}{
        \Return \textsc{Approved}\;
    }
    \uElseIf{\HasHardRejection{$v_*$}}{
        \Return \textsc{Rejected}\;
    }
    \Else{
        $\text{feedback} \leftarrow \ExtractFeedback{v_*}$\;
        $p \leftarrow \text{Architect.}\Revise{p, \text{feedback}}$\;
    }
}
\Return \textsc{NeedsHumanReview}\;
\end{algorithm}

% ============================================================================
% DEFINITION 1: Typed Migration Graph
% ============================================================================

\begin{definition}[Typed Migration Graph]
\label{def:tmg}
A Typed Migration Graph is a tuple $G = (V, E, \tau, \sigma)$ where:
\begin{itemize}
    \item $V$ is a finite set of vertices representing code artifacts;
    \item $E \subseteq V \times V$ is a set of directed edges representing dependencies;
    \item $\tau : V \rightarrow \mathcal{T}$ is a type function mapping vertices to types, where $\mathcal{T} = \{\textsc{Class}, \textsc{Interface}, \textsc{Method}, \textsc{Field}, \textsc{Import}, \textsc{Config}\}$;
    \item $\sigma : V \rightarrow \mathcal{S}$ is a state function mapping vertices to states, where $\mathcal{S} = \{\textsc{Unprocessed}, \textsc{Analyzed}, \textsc{Deprecated}, \textsc{Migrated}, \textsc{Failed}\}$.
\end{itemize}
\end{definition}


% ============================================================================
% PROPERTY 1: Safe Migration Order
% ============================================================================

\begin{property}[Safe Migration Order]
\label{prop:safe-order}
If $G$ is a directed acyclic graph, then processing vertices in topological order ensures that for any transition $\sigma(v) : \textsc{Deprecated} \rightarrow \textsc{Migrated}$, all dependencies $u$ where $(v, u) \in E$ satisfy $\sigma(u) = \textsc{Migrated}$.
\end{property}


% ============================================================================
% EQUATION: Consensus Scoring
% ============================================================================

\begin{equation}
\label{eq:consensus-score}
\text{score} = \frac{\sum_{a \in \mathcal{A}} w_a \cdot c_a \cdot \mathbb{1}[\text{approved}_a]}{|\mathcal{A}|} + \beta \cdot \mathbb{1}[\text{unanimous}]
\end{equation}

where $\mathcal{A} = \{\text{Archeologist}, \text{Architect}, \text{Validator}\}$, $w_a$ represents agent-specific weights, $c_a$ denotes confidence scores, and $\beta = 0.1$ provides a bonus for unanimous approval.
